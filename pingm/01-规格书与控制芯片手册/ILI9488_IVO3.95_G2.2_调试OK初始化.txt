Void ILI9488_EnterSleep_Code(void)
{
LCD_ILI9488_INDEX(0x28); //Display off
Delayms(10);
LCD_ILI9488_INDEX(0x10); //Sleep in
Delayms(120);
}

Void ILI9488_ExitSleep_Code(void)
{
LCD_ILI9488_INDEX(0x11);
Delayms(120);
LCD_ILI9488_INDEX(0x29);
Delayms(50);
}
